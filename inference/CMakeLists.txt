cmake_minimum_required(VERSION 3.20)
project(muse_inference VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fexceptions -frtti")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
option(MUSE_VULKAN "Build real Vulkan offload (native uses pinned RADV)" OFF)
option(MUSE_NEURAL_TTS "Build Qwen3-TTS through mtmd" OFF)
set(GGML_VULKAN ${MUSE_VULKAN} CACHE BOOL "" FORCE)
set(GGML_AVX ON CACHE BOOL "" FORCE)
set(GGML_AVX2 ON CACHE BOOL "" FORCE)
set(GGML_FMA ON CACHE BOOL "" FORCE)
set(GGML_F16C ON CACHE BOOL "" FORCE)
set(GGML_BMI2 ON CACHE BOOL "" FORCE)
set(GGML_AVX512 OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ${MUSE_NEURAL_TTS} CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/llama.cpp)
if(MUSE_NEURAL_TTS)
  # Keep the pinned implementation intact while exposing live audio-generation
  # buffer placement, which the public no-allocation probe does not report.
  set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp/tools/mtmd/mtmd.cpp"
    TARGET_DIRECTORY mtmd PROPERTIES HEADER_FILE_ONLY TRUE)
  target_sources(mtmd PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/mtmd_with_status.cpp")
endif()

# Use llama.cpp's single ggml ABI, following PSllama's speech port. Linking a
# second whisper ggml copy would create incompatible global/backend state.
set(WHISPER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/whisper.cpp")
add_library(muse_whisper STATIC
  whisper_with_status.cpp
  "${WHISPER_SOURCE}/src/aneforge/whisper-aneforge.cpp")
target_compile_definitions(muse_whisper PRIVATE WHISPER_VERSION="d1be6fde11ac")
target_include_directories(muse_whisper PUBLIC "${WHISPER_SOURCE}/include" PRIVATE "${WHISPER_SOURCE}/src")
target_link_libraries(muse_whisper PUBLIC ggml)

# eSpeak uses direct callback synthesis, without its playback device, async
# worker, MBROLA, sonic or external libraries. Real PCM is returned to the app.
set(ESPEAK_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/espeak-ng")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/espeak-config")
configure_file(espeak-config.h "${CMAKE_CURRENT_BINARY_DIR}/espeak-config/config.h" COPYONLY)
set(espeak_units common mnemonics error ieee80 compiledata compiledict dictionary encoding
  intonation langopts numbers phoneme phonemelist readclause setlengths soundicon spect
  ssml synthdata synthesize tr_languages translate translateword voices wavegen speech espeak_api klatt)
set(espeak_sources)
foreach(unit IN LISTS espeak_units)
  list(APPEND espeak_sources "${ESPEAK_SOURCE}/src/libespeak-ng/${unit}.c")
endforeach()
foreach(unit case categories ctype proplist scripts tostring)
  list(APPEND espeak_sources "${ESPEAK_SOURCE}/src/ucd-tools/src/${unit}.c")
endforeach()
add_library(muse_espeak STATIC ${espeak_sources})
target_compile_definitions(muse_espeak PRIVATE LIBESPEAK_NG_EXPORT=1 PATH_ESPEAK_DATA="")
target_include_directories(muse_espeak PUBLIC "${ESPEAK_SOURCE}/src/include"
  PRIVATE "${ESPEAK_SOURCE}/src/include/compat" "${ESPEAK_SOURCE}/src/ucd-tools/src/include"
  "${CMAKE_CURRENT_BINARY_DIR}/espeak-config")
add_library(muse_inference STATIC muse_inference.cpp)
if(CMAKE_CROSSCOMPILING)
  if(MUSE_VULKAN)
    target_sources(muse_inference PRIVATE gpu/native_vulkan.cpp gpu/compat.c)
    target_include_directories(muse_inference PRIVATE "${Vulkan_INCLUDE_DIR}")
  else()
    target_sources(muse_inference PRIVATE ps5_compat.c ps5_locale.c ps5_mbsnrtowcs.c ps5_wcsnrtombs.c)
  endif()
  target_compile_definitions(muse_inference PRIVATE MUSE_NATIVE_TITLE=1)
endif()
if(MUSE_VULKAN)
  target_compile_definitions(muse_inference PRIVATE MUSE_VULKAN=1)
endif()
if(MUSE_NEURAL_TTS)
  target_sources(muse_inference PRIVATE muse_neural_tts.cpp)
  target_compile_definitions(muse_inference PRIVATE MUSE_NEURAL_TTS=1)
  target_link_libraries(muse_inference PUBLIC mtmd)
endif()
target_include_directories(muse_inference PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(muse_inference PUBLIC llama muse_whisper muse_espeak)

# Optional Linux-only executable is for Docker CPU validation, never labeled as
# console execution. The same wrapper and model files are used by the title.
if(NOT CMAKE_CROSSCOMPILING)
  target_link_libraries(muse_espeak PUBLIC m)
  add_executable(muse-inference-check check.cpp)
  target_link_libraries(muse-inference-check PRIVATE muse_inference)
endif()

if(NOT CMAKE_CROSSCOMPILING AND MUSE_NEURAL_TTS)
  add_executable(muse-neural-check neural_check.cpp)
  target_link_libraries(muse-neural-check PRIVATE muse_inference)
endif()

option(MUSE_CODEGEN_CHECK "Build real-model Lua generation/repair harness on Linux" OFF)
if(NOT CMAKE_CROSSCOMPILING AND MUSE_CODEGEN_CHECK)
  file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../native/lua/sources.txt" muse_lua_units)
  set(muse_lua_sources)
  foreach(unit IN LISTS muse_lua_units)
    if(NOT unit STREQUAL "" AND NOT unit MATCHES "^#")
      list(APPEND muse_lua_sources "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/lua/src/${unit}")
    endif()
  endforeach()
  add_library(muse_codegen_lua STATIC ${muse_lua_sources} ../native/src/local_app.c)
  target_include_directories(muse_codegen_lua PUBLIC ../vendor/lua/src ../native/src)
  target_compile_definitions(muse_codegen_lua PRIVATE LUA_USE_C89)
  target_link_libraries(muse_codegen_lua PUBLIC m)
  add_executable(muse-codegen-check codegen_check.cpp)
  target_link_libraries(muse-codegen-check PRIVATE muse_inference muse_codegen_lua)
endif()
